// LISTENER EN TIEMPO REAL (CITAS Y CUENTAS) function initRealtimeListener() { if (isFirebaseActive && firebase.database) { // Citas en tiempo real const dbRefCitas = firebase.database().ref('citas'); dbRefCitas.on('child_added', (snapshot) => { const newApp = snapshot.val(); if (newApp && !appointments.some(a => a.id === newApp.id)) { appointments.push(newApp); localStorage.setItem('spa_appointments', JSON.stringify(appointments)); renderAdminAppointments(); checkAdminAlerts(); triggerRealtimePopup(newApp); } }); // Cuentas de pago en tiempo real (Escucha directa a la nube) const dbRefCuentas = firebase.database().ref('cuentas'); dbRefCuentas.on('value', (snapshot) => { const data = snapshot.val(); if (data) { paymentAccounts = Object.values(data); localStorage.setItem('spa_payment_accounts', JSON.stringify(paymentAccounts)); } else { paymentAccounts = []; localStorage.removeItem('spa_payment_accounts'); } renderClientPaymentAccounts(); renderAdminAccountsList(); }); } }
